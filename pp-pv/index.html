Header Expiry Badge — Drop-in Patch

Three small edits to your index_10_05_26_2215.html file. The badge will appear in the app header next to the theme picker / Firebase status pill, showing the customer's license expiry with a live, color-coded countdown.

0 What it looks like

Three states, depending on days remaining. The pill auto-switches between them every minute.

100+ days left
License:100d
30 days or less
License:18d
Expired
License:Expired

1 Add CSS   Append to your <style> block

Anywhere inside the existing <style>...</style> tag (your file already has hundreds of rules, just paste this at the end of it).

/* ── License expiry pill (header) ─────────────────────── */ .lic-pill { display:inline-flex; align-items:center; gap:8px; padding:6px 12px; margin-left:8px; border-radius:999px; font-size:12px; font-weight:600; background:var(--green-bg); border:1px solid var(--green); color:var(--green); letter-spacing:.01em; font-family:var(--font); cursor:default; } .lic-pill.warn { background:var(--orange-bg); border-color:var(--orange); color:var(--orange); } .lic-pill.danger { background:var(--red-bg); border-color:var(--red); color:var(--red); } .lic-pill .lp-dot { width:7px; height:7px; border-radius:50%; background:currentColor; animation: lpPulse 2s infinite; } .lic-pill .lp-lbl { opacity:.85; font-weight:500; } @keyframes lpPulse { 0% { box-shadow:0 0 0 0 currentColor; } 70% { box-shadow:0 0 0 6px transparent; } 100% { box-shadow:0 0 0 0 transparent; } }
Why var(--green) etc.? Your existing CSS already defines colour tokens for every theme (Midnight Gold, Ocean Depth, Emerald Forest, …). The pill will automatically retheme when the user picks a different theme — no extra work.

2 Add HTML   Insert the pill in your header

In index_10_05_26_2215.html, find this line (around line 1055):

<!-- Developer shortcut button: only shown when logged in as 'developer' --> <button id="devPanelBtn" onclick="openDevPanel()"

Insert this snippet directly before that comment:

<!-- License expiry pill --> <div class="lic-pill" id="licExpiryPill" style="display:none" title="Click for license details" onclick="showLicenseInfo()"> <span class="lp-dot"></span> <span class="lp-lbl">License:</span> <span id="licExpiryText">—</span> </div>

3 Add JS   Append to your main <script> block

Paste this anywhere inside your existing main script tag. It reads the activated license key from localStorage, checks Firestore (if available) or local storage, then renders the pill.

/* ══════════════════════════════════════════════════════════ LICENSE EXPIRY BADGE — header pill, live countdown ══════════════════════════════════════════════════════════ */ let _licInfo = null; // cached license object let _licTickerHandle = null; // setInterval handle async function initLicenseBadge() { // Only the superadmin sees the pill. if (!currentUser || currentUser.access !== 'admin') return; const key = localStorage.getItem('pv_license'); if (!key) { // No license activated yet — show a "Not Licensed" pill in red. _licInfo = { status:'unlicensed' }; renderLicenseBadge(); return; } try { // Read from Firestore if available, fall back to local panel storage. if (typeof db !== 'undefined') { const doc = await db.collection('licenses').doc(key).get(); if (doc.exists) _licInfo = doc.data(); } else { const raw = localStorage.getItem('pv_licenses_v1'); const arr = raw ? JSON.parse(raw) : []; _licInfo = arr.find(l => l.key === key) || null; } } catch (e) { console.warn('License lookup failed:', e); } renderLicenseBadge(); // Tick every minute so the day counter stays current. if (_licTickerHandle) clearInterval(_licTickerHandle); _licTickerHandle = setInterval(renderLicenseBadge, 60000); } function renderLicenseBadge() { const pill = document.getElementById('licExpiryPill'); const txt = document.getElementById('licExpiryText'); if (!pill || !txt) return; if (!_licInfo || _licInfo.status === 'unlicensed') { pill.style.display = 'inline-flex'; pill.className = 'lic-pill danger'; txt.textContent = 'Not Licensed'; return; } if (_licInfo.status === 'revoked') { pill.style.display = 'inline-flex'; pill.className = 'lic-pill danger'; txt.textContent = 'Revoked'; return; } const days = Math.ceil((_licInfo.expiresAt - Date.now()) / 86400000); pill.style.display = 'inline-flex'; if (days < 0) { pill.className = 'lic-pill danger'; txt.textContent = 'Expired'; } else if (days <= 30) { pill.className = 'lic-pill warn'; txt.textContent = days + 'd'; } else { pill.className = 'lic-pill'; txt.textContent = days + 'd'; } } function showLicenseInfo() { if (!_licInfo) { toast('No License', 'This device is not yet licensed. Contact the developer.', true); return; } const days = Math.ceil((_licInfo.expiresAt - Date.now()) / 86400000); const dateStr = new Date(_licInfo.expiresAt).toLocaleDateString(); const msg = 'Key: ' + _licInfo.key + '\nExpires: ' + dateStr + '\nDays left: ' + (days < 0 ? 'Expired '+Math.abs(days)+'d ago' : days) + '\nActivations: ' + (_licInfo.activationCount || 0) + '/' + (_licInfo.maxActivations || '∞'); toast('License Status', msg); } // Hook into the existing post-login flow. // Find your existing login success function (where currentUser is set // and updateBadges() / initNotifBell() are called) and add this line: // initLicenseBadge();
Where to call initLicenseBadge(): in your existing code, search for initNotifBell(). Add initLicenseBadge(); on the line right after it — both follow the same pattern of running after the user is authenticated.

4 Optional   Block the app on expiry

If the developer panel's "Block on expiry" toggle is on, the customer device should refuse to load. Add this check inside your existing login function, right before the app chrome is shown:

if (_licInfo && (_licInfo.status === 'revoked' || _licInfo.expiresAt < Date.now())) { document.body.innerHTML = ` <div style="display:grid;place-items:center;height:100vh; font-family:DM Sans,sans-serif;color:#f87171; background:#0a0a18;padding:20px;text-align:center"> <div> <div style="font-size:48px;margin-bottom:14px">🔒</div> <h1 style="font-size:22px;margin-bottom:8px;color:#ececf5"> License Expired </h1> <p style="color:#7a7a99;max-width:420px"> This installation of PlantVisit is no longer authorized. Please contact your administrator to renew the license. </p> </div> </div>`; return; }
